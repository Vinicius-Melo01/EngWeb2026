# queries.txt
# DB: jogostabuleiro

# 1. Quantos jogos estão registados na base de dados?
db.jogos.countDocuments()

# 2. Quantos jogos pertencem à categoria "Family"?
db.jogos.countDocuments({ category: "Family" })

# 3. Lista de autores (ordenada alfabeticamente, sem repetições)
db.jogos.aggregate([
  { $unwind: "$autores" },
  { $group: { _id: "$autores.name" } },
  { $sort: { _id: 1 } }
])

# 4. Distribuição de jogos por ano de lançamento
db.jogos.aggregate([
  { $group: { _id: "$year", total: { $sum: 1 } } },
  { $sort: { _id: 1 } }
])

# 5. Distribuição de jogos por editora
db.jogos.aggregate([
  { $unwind: "$editoras" },
  { $group: { _id: "$editoras.name", total: { $sum: 1 } } },
  { $sort: { total: -1 } }
])
